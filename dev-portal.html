async function approvePatrol(key){ try{ await fetch('/api/approve',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({phone:key})}); }catch(e){} // keep localStorage in sync for instant UI if(usersData[key]) usersData[key].status='active'; localStorage.setItem('zondi_users',JSON.stringify(usersData)); render(); } async function deletePatrol(key){ if(!confirm('Delete '+key+'?')) return; try{ await fetch('/api/delete',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({phone:key})}); }catch(e){} delete usersData[key]; localStorage.setItem('zondi_users',JSON.stringify(usersData)); render(); closeFile(); } function deleteFile(){ deletePatrol(currentKey); } async function saveEdit(){ let old_phone=currentKey; let newName=document.getElementById('editName').value.trim(); let newId=document.getElementById('editId').value.trim(); let newRole=document.getElementById('editRole').value; let newPhone=document.getElementById('editPhone').value.trim(); try{ await fetch('/api/edit',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({old_phone,new_phone:newPhone,name:newName,client_id:newId,role:newRole})}); }catch(e){} // local sync let u=usersData[old_phone]; if(!u) return; u.name=newName||u.name; u.client_id=newId||u.client_id; u.role=newRole; if(newPhone && newPhone!==old_phone){ usersData[newPhone]=u; delete usersData[old_phone]; currentKey=newPhone; } localStorage.setItem('zondi_users',JSON.stringify(usersData)); render(); alert('✅ Saved'); openFile(currentKey); }
